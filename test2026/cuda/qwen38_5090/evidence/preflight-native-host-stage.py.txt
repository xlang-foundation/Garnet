from pathlib import Path
import argparse
import hashlib
import json
import re
import shutil
import sys
import zipfile

root = Path('D:/CantorAI/5090-study-20261023')
methods = root / 'Garnet/test2026/cuda/qwen38_5090'
sys.path.insert(0, str(methods))
from run_bf16_low_batch_native_build import idle
from run_sglang_observed_fixed4k import closed_native_full4k
sha = lambda path: hashlib.sha256(path.read_bytes()).hexdigest()
parser = argparse.ArgumentParser()
parser.add_argument('--closed-reference-job', required=True)
parser.add_argument('--closed-reference-archive-sha256', required=True)
args = parser.parse_args()
receipt = root / 'results/native-host-stage-preflight.json'
assert not receipt.exists()
name = args.closed_reference_job
digest = args.closed_reference_archive_sha256
assert re.fullmatch(r'sglang-observed-fixed4k-ordinary-attempt[1-9][0-9]*', name)
assert re.fullmatch('[0-9a-f]{64}', digest)
archive = root / 'results' / (name + '-evidence.zip')
assert sha(archive) == digest
with zipfile.ZipFile(archive) as z:
    manifest = json.loads(z.read('evidence-manifest.json'))
    assert z.testzip() is None and len(z.namelist()) == len(set(z.namelist())) == len(manifest['files']) + 1
    for path, expected in manifest['files'].items():
        assert hashlib.sha256(z.read(path)).hexdigest() == expected
    closed = json.loads(z.read('results/' + name + '/workflow.json'))
assert closed == json.loads((root / 'results' / name / 'workflow.json').read_bytes())
assert type(closed['exit_code']) is int and type(closed['server_exit_code']) is int and closed['cleanup_finished'] is True
assert closed['source_sha256'] == closed['source_after_sha256'] and closed['runtime_dependencies'] == closed['runtime_dependencies_after']
proof_path = root / 'results' / (name + '-verified.json')
proof = json.loads(proof_path.read_bytes())
assert proof['archive_sha256'] == digest
assert proof['status'] == ('reference_fixed4k_observed_profile_replayed' if closed['exit_code'] == 0 else 'reference_fixed4k_observed_failure_preserved')
input_path = root / 'results/native-host-stage-preflight-input.json'
p = json.loads(input_path.read_bytes())
native = closed_native_full4k(root, p['full4k_parent']['archive_sha256'])
assert native['proof_sha256'] == p['full4k_parent']['proof_sha256']
idle()
targets, drift, runtime = {}, {}, {}
for name, expected in p['expected'].items():
    path = root / name
    assert path.resolve().is_relative_to(methods.resolve())
    exists = path.exists()
    actual = sha(path) if exists and path.is_file() else None
    targets[name] = {'exists': exists, 'sha256': actual, 'expected_sha256': expected, 'matches': exists and actual == expected}
for name, expected in p['existing'].items():
    path = root / name
    actual = sha(path) if path.is_file() else None
    if actual != expected:
        drift[name] = {'expected': expected, 'actual': actual}
for name, expected in p['runtime_dependencies'].items():
    actual = sha(root / 'build/bin' / name)
    if actual != expected:
        runtime[name] = {'expected': expected, 'actual': actual}
conflicts = {name: row for name, row in targets.items() if row['exists'] and not row['matches']}
passed = not drift and not runtime and not conflicts
result = {'status': 'native_host_stage_preflight_passed' if passed else 'native_host_stage_preflight_failed',
          'closed_reference_archive_sha256': digest, 'closed_reference_proof_sha256': sha(proof_path),
          'input_sha256': sha(input_path), 'targets': targets, 'existing_input_drift': drift,
          'runtime_drift': runtime, 'conflicts': conflicts, 'free_study_disk_bytes': shutil.disk_usage(root).free,
          'scope': 'Actual independently replayed closed reference/native archives, source/runtime and idle-process checks. Read-only except receipt; no model execution or source replacement.'}
receipt.write_text(json.dumps(result, indent=2), encoding='utf8')
print(json.dumps(result))
raise SystemExit(0 if passed else 1)
