from pathlib import Path
import datetime
import hashlib
import json
import subprocess
import sys

work = Path(__file__).resolve().parent
methods = Path('D:/CantorAI/work/qwen38-5090/test2026/cuda/qwen38_5090')
job = 'sglang-observed-fixed4k-ordinary-attempt1'
output = work.parent / 'outputs'
proof = json.loads((output / (job + '-remote-verified.json')).read_bytes())
assert hashlib.sha256((output / (job + '-evidence.zip')).read_bytes()).hexdigest() == proof['archive_sha256']
command = [sys.executable, str(methods / 'verify_sglang_observed_fixed4k.py'),
           '--proofs', str(output), '--job', job, '--corpus', str(work / 'mmmu/quality-corpus'), '--scratch', str(work)]
log_path = output / (job + '-local-replay.log')
with log_path.open('xb') as stream:
    code = subprocess.run(command, stdout=stream, stderr=subprocess.STDOUT).returncode
result = {'exit_code': code, 'argv': command,
          'utc': datetime.datetime.now(datetime.timezone.utc).isoformat(),
          'archive_sha256': proof['archive_sha256'], 'log_sha256': hashlib.sha256(log_path.read_bytes()).hexdigest()}
if code == 0:
    actual = json.loads((output / (job + '-verified.json')).read_bytes())
    assert actual == proof
    result.update(proof_pair_identical=True, status=actual['status'])
else:
    print(log_path.read_text(encoding='utf8', errors='replace')[-6000:])
path = work / 'sglang-observed-fixed4k-local-replay-receipt.json'
assert not path.exists()
path.write_text(json.dumps(result, indent=2), encoding='utf8')
print(json.dumps(result))
raise SystemExit(code)
