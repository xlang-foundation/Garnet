from pathlib import Path
import hashlib,json,zipfile
import numpy as np
work=Path(__file__).resolve().parent;out=work.parent/'outputs';job='bf16-batch8-external-reference-attempt1';parent='paged-batch8-model-preservation-attempt1'
sha=lambda raw:hashlib.sha256(raw).hexdigest()
archive=out/(job+'-evidence.zip');old=out/(parent+'-evidence.zip')
with zipfile.ZipFile(archive) as z,zipfile.ZipFile(old) as b:
    prefix='results/'+job+'/model/';previous='results/'+parent+'/model/'
    p=json.loads(z.read(prefix+'report.json'));q=json.loads(b.read(previous+'report.json'))
    assert len(p['cases'])==1
    c=p['cases'][0];o=q['cases'][c['baseline_case_index']];rows=c['physical_rows']
    assert c['inputs']==[o['inputs'][i] for i in rows]
    observations=[]
    for phase in ('reset_state','prefill_state','final_state'):
        a=c[phase];d=o[phase];components=[]
        assert len(a['logical']['components'])==len(d['logical']['components'])==128
        for left,right in zip(a['logical']['components'],d['logical']['components']):
            assert all(left[k]==right[k] for k in ('layer','kind','role','logical_words'))
            different=[r for r,i in enumerate(rows) if left['row_sha256'][r]!=right['row_sha256'][i]]
            if different:components.append({k:left[k] for k in ('layer','kind','role','logical_words')}|{'different_physical_rows':different})
        vectors={}
        for kind in ('hidden','normalized'):
            info=a['vectors'][kind];ref=d['vectors'][kind];raw=z.read(prefix+info['file']);refraw=b.read(previous+ref['file'])
            assert sha(raw)==info['sha256'] and sha(refraw)==ref['sha256'] and len(raw)==info['bytes']==81920 and len(refraw)==ref['bytes']==163840
            x=np.frombuffer(raw,dtype='<f4').reshape(4,5120).astype(np.float64);y=np.frombuffer(refraw,dtype='<f4').reshape(8,5120)[rows].astype(np.float64)
            assert np.isfinite(x).all() and np.isfinite(y).all()
            vectors[kind]=[{'physical_row':r,'equal_scalars':int(np.count_nonzero(left==right)),'scalars':5120,'max_absolute_difference':float(np.max(np.abs(left-right))),
                            'relative_l2_difference':float(np.linalg.norm(left-right)/max(np.linalg.norm(right),1e-300))} for r,(left,right) in enumerate(zip(x,y))]
        observations.append({'phase':phase,'slot_counts_match':a['logical']['slots']==[d['logical']['slots'][i] for i in rows],
            'logical_row_hashes_match':[h==d['logical']['logical_row_sha256'][i] for h,i in zip(a['logical']['logical_row_sha256'],rows)],
            'different_components':components,'vectors':vectors})
    ids=[]
    for row,i in enumerate(rows):
        actual=c['generated_ids'][row];expected=o['generated_ids'][i]
        assert len(actual)==len(expected)==8
        ids.append({'physical_row':row,'actual':actual,'expected':expected,'first_different_token_zero_based':next((t for t,(a,b) in enumerate(zip(actual,expected)) if a!=b),None)})
    workflow=json.loads(z.read('results/'+job+'/workflow.json'))
    assert workflow['exit_code']==1 and workflow['cleanup_finished'] is True and workflow['source_sha256']==workflow['source_after_sha256'] and workflow['runtime_dependencies']==workflow['runtime_dependencies_after'] and workflow['environment_before']==workflow['environment_after']
r={'status':'actual_b8_b4_strict_failure_boundaries_analyzed','archive_sha256':sha(archive.read_bytes()),'parent_archive_sha256':sha(old.read_bytes()),'completed_cases':1,'planned_cases':18,'group':c['group'],'half':c['half'],'prompt_lengths':[len(v['prompt_ids']) for v in c['inputs']],'actual_inputs_exact':True,'generated_ids':ids,'phase_observations':observations,'source_runtime_sdk_preserved':True,'cause_established':False,'harmlessness_or_equivalence_proven':False,'performance_admitted':False,'two_times_target_achieved':False,'scope':'Only actual first completed four-row short case. Complete expanded logical-state hashes identify equality/difference; raw caches are not present. Raw hidden/normalized FP32 vectors permit exact scalar/norm differences. No per-layer arithmetic cause or numerical tolerance admission inferred.'}
(out/'Garnet-B8-B4-Reference-Failure-Analysis.json').write_bytes(json.dumps(r,indent=2).encode())
print(json.dumps({'status':r['status'],'prompt_lengths':r['prompt_lengths'],'first_token_differences':[v['first_different_token_zero_based'] for v in ids],
                  'phases':[{'phase':v['phase'],'different_components':len(v['different_components']),'first_component':v['different_components'][0] if v['different_components'] else None,'hidden_max_abs':[x['max_absolute_difference'] for x in v['vectors']['hidden']]} for v in observations]}))
