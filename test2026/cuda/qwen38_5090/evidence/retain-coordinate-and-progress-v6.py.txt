"""Retain verified reporting inputs and append paper logs without rewriting history."""
from pathlib import Path
import hashlib
import json
import zipfile

W = Path(__file__).resolve().parent
O = W.parent / 'outputs'
R = Path('D:/CantorAI/work/qwen38-5090')
E = R / 'test2026/cuda/qwen38_5090/evidence'
F = E / 'figures'
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
owned = []
retained = {}


def keep(dest, raw):
    assert not dest.exists(), dest
    dest.write_bytes(raw)
    assert dest.read_bytes() == raw
    owned.append(dest.relative_to(R).as_posix())
    retained[dest.relative_to(E).as_posix()] = sha(dest)


J = 'key35-coordinate-trajectory-attempt1'
proof = O / (J + '-verified.json')
q = json.loads(proof.read_bytes())
assert proof.read_bytes() == (O / (J + '-remote-verified.json')).read_bytes()
assert q['exit_code'] == 0
archive = O / (J + '-evidence.zip')
assert sha(archive) == q['archive_sha256']
summary = json.loads((O / 'Garnet-Key35-Coordinate-Trajectory-Progress.json').read_bytes())
assert summary['first_observed_selected_value_difference_step'] == 45
assert summary['CPU_controls'] == 52
assert summary['same_profile_full_cache_bytes_reproduced']
assert summary['baseline_source_runtime_SDK_config_index_CMake_all34_restored']
assert not summary['two_times_target_achieved']
for name in (proof.name, J + '-remote-verified.json', 'Garnet-Key35-Coordinate-Trajectory-Progress.json'):
    keep(E / name, (O / name).read_bytes())
with zipfile.ZipFile(archive) as z:
    mf = json.loads(z.read('evidence-manifest.json'))
    base = 'results/' + J + '/'
    keep(E / 'key35-coordinate-trajectory-model1-manifest.json', z.read('evidence-manifest.json'))
    for name in mf['files']:
        if name.startswith(base) and name.endswith('.json'):
            keep(E / ('key35-coordinate-trajectory-model1-' + name.removeprefix(base).replace('/', '-')), z.read(name))
    for phase in ('cpu', 'reference', 'candidate'):
        for ext in ('log', 'jsonl'):
            keep(E / ('key35-coordinate-trajectory-model1-' + phase + '-commands.' + ext), z.read(base + phase + '-ledger/commands.' + ext))
for name in ('commands.log', 'commands.jsonl'):
    keep(E / ('key35-coordinate-trajectory-model1-local-replay-' + name), (W / 'key35-coordinate-trajectory-local-replay' / name).read_bytes())
for name in ('postprocess-key35-coordinate-trajectory.py', 'replay-key35-coordinate-trajectory.py', Path(__file__).name):
    keep(E / (name + '.txt'), (W / name).read_bytes())

for stem, generator in (
    ('Garnet-Key35-Coordinate-Trajectory', 'render-key35-coordinate-trajectory.py'),
    ('Garnet-Progress-Comparison-2026-10-27-v6', 'render-progress-comparison-20261027-v6.py'),
):
    receipt_path = O / (stem + '-Receipt.json')
    receipt = json.loads(receipt_path.read_bytes())
    assert sha(W / generator) == receipt['generator_sha256']
    for name, digest in receipt['artifacts'].items():
        assert sha(O / name) == digest
        keep(F / name, (O / name).read_bytes())
    keep(F / receipt_path.name, receipt_path.read_bytes())
    keep(F / (generator + '.txt'), (W / generator).read_bytes())
    for name, digest in receipt.get('input_sha256', {}).items():
        p = Path(name)
        assert sha(p) == digest
        if p.suffix == '.json':
            keep(F / ('progress-v6-input-' + p.name), p.read_bytes())
keep(F / 'prepare-progress-comparison-v6.py.txt', (W / 'prepare-progress-comparison-v6.py').read_bytes())
record = {
    'status': 'ordinary_coordinate_timeline_and_progress_v6_retained',
    'archive_sha256': sha(archive), 'archive_bytes': archive.stat().st_size,
    'proof_sha256': sha(proof), 'retained_sha256': dict(retained),
    'paired_remote_local_proof_bytes_identical': True,
    'CPU_controls': 52,
    'first_observed_sampled_coordinate_difference_step': 45,
    'first_global_divergence_or_unique_cause_proved': False,
    'baseline_restored': True,
    'new_benchmark_executed_for_report': False,
    'overall_two_times_target_achieved': False,
    'visual_QA': 'Both PNGs inspected: labels, table and step transitions visible.',
    'raw_archive_retained_in_outputs_and_remote_results': True,
    'remote_CPython_NumPy': '3.12.3 / 2.4.4',
    'local_CPython_NumPy': '3.12.10 / 1.26.4',
    'downloaded_native_artifact_executed_locally': False,
}
keep(E / 'key35-coordinate-trajectory-model1-retention.json', json.dumps(record, indent=2).encode())
entry = ('\n\n2026-10-27 ordinary key35 coordinate timeline and progress comparison v6: '
         'fresh source60 B1 and source85 flagoff B4 ordinary runs completed. '

         'remote/local raw replays byte-identical, proof '+sha(proof)+'. '
         'Raw archive '+sha(archive)+' ('+str(archive.stat().st_size)+'B). '
         'All1861 owned files/runtime/SDK/config+index/CMake/current34objects/'
         'baselineDLL restored and owned SSD nonces retired. No native/graph/'
         'kernel/precision changes in the diagnostic. Historical full4K/cache62 '

for rel in ('docs/research/qwen38-5090-development-log.md', 'docs/research/qwen38-5090-implementation-task.md'):
    p = R / rel
    before = p.read_bytes()
    assert b'ordinary key35 coordinate timeline and progress comparison v6:' not in before
