from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
from datetime import datetime, timezone
import ast, hashlib, json, urllib.request

W=Path(__file__).resolve().parent
OUT=W/'upstream-reference-guidance'
OUT.mkdir(exist_ok=True)
SOURCES={
 'vllm-0.32.0-attention-backend.py':'https://raw.githubusercontent.com/vllm-project/vllm/v0.32.0/vllm/v1/attention/backend.py',
 'vllm-0.32.0-gdn-attn.py':'https://raw.githubusercontent.com/vllm-project/vllm/v0.32.0/vllm/v1/attention/backends/gdn_attn.py',
 'sglang-0.5.22-server-args.py':'https://raw.githubusercontent.com/sgl-project/sglang/v0.5.22/python/sglang/srt/server_args.py',
 'sglang-0.5.22-qwen3-5.py':'https://raw.githubusercontent.com/sgl-project/sglang/v0.5.22/python/sglang/srt/models/qwen3_5.py',
}
def fetch(item):
 name,url=item
 try:
  with urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'CantorAI-RTX5090-study'}),timeout=45) as r:
   raw=r.read();final=r.url
  ast.parse(raw)
  p=OUT/name
  if p.exists():assert p.read_bytes()==raw
  else:p.write_bytes(raw)
  return name,dict(url=url,resolved_url=final,sha256=hashlib.sha256(raw).hexdigest(),bytes=len(raw),success=True)
 except Exception as e:return name,dict(url=url,success=False,error=type(e).__name__+': '+str(e))
rows=dict(ThreadPoolExecutor(max_workers=4).map(fetch,SOURCES.items()))
result=dict(retrieved_utc=datetime.now(timezone.utc).isoformat(),scope='Official upstream version tags only; installed package identity and actual runtime compatibility remain to be verified. Read-only CPU/network; no remote source mutation or GPU/model run.',sources=rows)
if all(r['success'] for r in rows.values()):
 base=ast.parse((OUT/'vllm-0.32.0-attention-backend.py').read_bytes())
 gdn=ast.parse((OUT/'vllm-0.32.0-gdn-attn.py').read_bytes())
 cls=next(n for n in base.body if isinstance(n,ast.ClassDef) and n.name=='AttentionBackend')
 method=next(n for n in cls.body if isinstance(n,ast.FunctionDef) and n.name=='supports_batch_invariance')
 assert len(method.body)==1 and isinstance(method.body[0],ast.Return) and method.body[0].value.value is False
 gcls=next(n for n in gdn.body if isinstance(n,ast.ClassDef) and n.name=='GDNAttentionBackend')
 assert [ast.unparse(n) for n in gcls.bases]==['AttentionBackend']
 assert not any(isinstance(n,ast.FunctionDef) and n.name=='supports_batch_invariance' for n in gcls.body)
 result['vllm_upstream_tag_batch_invariance']=dict(GDN_backend_inherits_default_false=True,default_method_line=method.lineno,GDN_class_line=gcls.lineno,installed_code_or_model_runtime_claim=False)
receipt=OUT/'retrieval-receipt.json'
assert not receipt.exists()
receipt.write_bytes(json.dumps(result,indent=2).encode())
print(json.dumps(result))
