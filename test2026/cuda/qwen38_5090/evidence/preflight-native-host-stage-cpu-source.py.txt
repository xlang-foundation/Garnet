from pathlib import Path
import ast
import hashlib
import json
import shutil
import sys
import zipfile

root = Path('D:/CantorAI/5090-study-20261023')
methods = root / 'Garnet/test2026/cuda/qwen38_5090'
sys.path.insert(0, str(methods))
from run_bf16_low_batch_native_build import idle
sha = lambda path: hashlib.sha256(path.read_bytes()).hexdigest()
receipt = root / 'results/native-host-stage-cpu-source-preflight.json'
assert not receipt.exists()
input_path = root / 'results/native-host-stage-cpu-source-preflight-input.json'
p = json.loads(input_path.read_bytes())
name = 'native-host-stage-full4k-attempt1'
archive = root / 'results' / (name + '-evidence.zip')
with zipfile.ZipFile(archive) as z:
    manifest = json.loads(z.read('evidence-manifest.json'))
    assert z.testzip() is None and len(z.namelist()) == len(set(z.namelist())) == len(manifest['files']) + 1
    for path, expected in manifest['files'].items(): assert hashlib.sha256(z.read(path)).hexdigest() == expected
    w = json.loads(z.read('results/' + name + '/workflow.json'))
    prep_raw = z.read('Garnet/test2026/cuda/qwen38_5090/evidence/native-host-stage-run-preparation.json')
    original = json.loads(prep_raw)
    assert hashlib.sha256(prep_raw).hexdigest() == w['preparation_sha256'] == p['original_preparation_sha256']
    log = z.read('results/' + name + '/cpu-ledger/commands.log')
    assert b'Ran 27 tests' in log and b'\nOK' in log and b'FAILED' not in log
assert w == json.loads((root / 'results' / name / 'workflow.json').read_bytes())
assert type(w['exit_code']) is int and w['exit_code'] == 0 and w['cleanup_finished'] is True and w['library_unchanged'] is True
assert w['source_sha256'] == w['source_after_sha256'] == original['files']
assert w['runtime_dependencies'] == w['runtime_dependencies_after'] == original['runtime_dependencies']
assert original['expected_cpu_test_count'] == 29 and 'Garnet/test2026/cuda/qwen38_5090/test_native_worker_io.py' not in original['files']
idle()
targets, drift, runtime = {}, {}, {}
for name, expected in p['expected'].items():
    path = root / name; assert path.resolve().is_relative_to(methods.resolve())
    exists = path.exists(); actual = sha(path) if exists and path.is_file() else None
    targets[name] = {'exists': exists, 'sha256': actual, 'expected_sha256': expected, 'matches': exists and actual == expected}
for name, expected in original['files'].items():
    path = root / name; actual = sha(path) if path.is_file() else None
    if actual != expected: drift[name] = {'expected': expected, 'actual': actual}
for name, expected in original['runtime_dependencies'].items():
    actual = sha(root / 'build/bin' / name)
    if actual != expected: runtime[name] = {'expected': expected, 'actual': actual}
conflicts = {name: row for name, row in targets.items() if row['exists'] and not row['matches']}
passed = not drift and not runtime and not conflicts
io_source = methods / 'test_native_worker_io.py'
io_methods = [node.name for node in ast.walk(ast.parse(io_source.read_bytes())) if isinstance(node, ast.FunctionDef) and node.name.startswith('test_')]
result = {'status': 'native_host_stage_cpu_source_preflight_passed' if passed else 'native_host_stage_cpu_source_preflight_failed',
          'original_archive_sha256': sha(archive), 'original_preparation_sha256': w['preparation_sha256'],
          'input_sha256': sha(input_path), 'targets': targets, 'existing_input_drift': drift,
          'runtime_drift': runtime, 'conflicts': conflicts, 'free_study_disk_bytes': shutil.disk_usage(root).free,
          'original_io_test_postrun_sha256': sha(io_source), 'original_io_test_postrun_methods': io_methods,
          'scope': 'Closed original successful model job, actual27CPU count and missing29metadata IO test-source pin; source/runtime/idle checks. Old IO source observed post-run only. Read-only except receipt; no model, GPU or source replacement.'}
receipt.write_text(json.dumps(result, indent=2), encoding='utf8')
print(json.dumps(result)); raise SystemExit(0 if passed else 1)
