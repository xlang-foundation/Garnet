def method(raw,cls,name):
    c=next(n for n in ast.parse(raw).body if isinstance(n,ast.ClassDef) and n.name==cls)
    return copy.deepcopy(next(n for n in c.body if isinstance(n,ast.FunctionDef) and n.name==name))

# Remove only the three explicit observer operations; all original arithmetic,
# state layouts, call arguments and commit payload construction must remain.
removed=[]
class Strip(ast.NodeTransformer):
    def visit_Assign(self,n):
        if any(isinstance(t,ast.Name) and t.id=='boundary_outputs' for t in n.targets):removed.append('initialize');return None
        return self.generic_visit(n)
    def visit_If(self,n):
        if ast.unparse(n.test)=="layer['index'] in self.boundary_layers":removed.append('append');return None
        return self.generic_visit(n)
    def visit_BinOp(self,n):
        if isinstance(n.op,ast.Add) and isinstance(n.right,ast.Name) and n.right.id=='boundary_outputs':removed.append('return_suffix');return self.visit(n.left)
        return self.generic_visit(n)
