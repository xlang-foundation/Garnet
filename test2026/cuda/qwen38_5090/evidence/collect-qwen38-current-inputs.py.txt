"""Fingerprint current inference inputs after the closed lifecycle run, CPU only."""
from pathlib import Path
import datetime,hashlib,json,sys
ROOT=Path('D:/CantorAI/5090-study-20261023');M=ROOT/'Garnet/test2026/cuda/qwen38_5090';sys.path.insert(0,str(M))
from run_bf16_low_batch_native_build import idle
def sha_file(path):
    h=hashlib.sha256()
    with path.open('rb') as f:
        while block:=f.read(8*1024*1024):h.update(block)
    return h.hexdigest()

def main():
    idle();job='b4-padding-lifecycle-attempt1';proof=ROOT/'results'/(job+'-verified.json');q=json.loads(proof.read_bytes())
    assert q['exit_code']==0 and q['B4_padding_lifecycle_case_gate_passed'] and q['source_runtime_SDK_config_index_CMake_all34_baseline_restored']
    p=json.loads((M/'evidence/b4-padding-lifecycle-preparation.json').read_bytes())
    for n,h in p['checkpoints'].items():assert sha_file(ROOT/n)==h
    directory=ROOT/'models/Qwen3.8-27B-NVFP4';index=json.loads((directory/'model.safetensors.index.json').read_bytes());weights=set(index['weight_map'].values())
    assert weights and all(not Path(n).is_absolute() and '..' not in Path(n).parts for n in weights)
    paths={n for n in directory.rglob('*') if n.is_file() and not any(part.startswith('.') for part in n.relative_to(directory).parts) and n.suffix in ('.json','.safetensors','.model','.txt','.jinja','.tiktoken')}
    paths|={directory/n for n in weights};assert all(n.is_file() for n in paths)
    files={};stats={}
    for path in sorted(paths):
        before=path.stat();digest=sha_file(path);after=path.stat();assert (before.st_size,before.st_mtime_ns)==(after.st_size,after.st_mtime_ns),('Input changed while hashing',path.name)
        key=path.relative_to(ROOT).as_posix();files[key]=digest;stats[key]=dict(bytes=after.st_size,mtime_ns=after.st_mtime_ns)
        print(json.dumps(dict(file=key,bytes=after.st_size,sha256=digest)),flush=True)
    idle();target=ROOT/'results/qwen38-nvfp4-current-inputs.json'
    doc=dict(status='current_inference_inputs_fingerprinted_CPU_only',revision='482ca0f3832238542f8f5295dde86b5f22711d80',files=files,file_stats=stats,generated_UTC=datetime.datetime.now(datetime.timezone.utc).isoformat(),collector_sha256=sha_file(Path(__file__)),closed_parent=job,parent_proof_sha256=sha_file(proof),config_index_pins_matched=True,posthoc_relative_to_parent=True,historical_parent_shard_identity_retroactively_proved=False,full_HF_LFS_OID_comparison_performed=False,model_or_GPU_execution_performed=False,scope='All indexed weight shards and visible JSON/tokenizer/model/text/template inference files. Current full hashes for future before/after measurements. Historical runs pinned config/index, so this posthoc manifest does not retroactively establish their shard-file hashes. Local hashes are not a fresh full HF LFS OID verification.')
    if target.exists():
        old=json.loads(target.read_bytes());assert old['files']==files and old['revision']==doc['revision'];status='existing_manifest_rechecked'
    else:target.write_bytes(json.dumps(doc,indent=2).encode());status=doc['status']
    print(json.dumps(dict(status=status,manifest_sha256=sha_file(target),file_count=len(files),weight_shards=len(weights),inference_bytes=sum(s['bytes'] for s in stats.values()),posthoc_not_retroactive=True)),flush=True)

if __name__=='__main__':main()
