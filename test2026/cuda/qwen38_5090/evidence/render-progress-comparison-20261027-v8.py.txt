def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def read(path):
    inputs[str(path)] = sha(path)
    return json.loads(path.read_bytes())

def closed_update(name, job):
    d = read(O / name)
    proof = O / (job + '-verified.json')
    remote = O / (job + '-remote-verified.json')
    archive = O / (job + '-evidence.zip')
    assert proof.read_bytes() == remote.read_bytes()
    assert sha(proof) == d['proof_sha256'] and sha(archive) == d['archive_sha256']
    assert d['paired_remote_local_proof_bytes_identical'] and not d['two_times_target_achieved']
    q = json.loads(proof.read_bytes())
    assert q['exit_code'] == d['exit_code']
    for p in (proof, remote, archive):
        inputs[str(p)] = sha(p)
    return d
