inputs = {}


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def read(path):
    inputs[str(path)] = sha(path)
    return json.loads(path.read_bytes())


receipt = read(O / (BASE + '-Receipt.json'))
data = read(O / (BASE + '-Data.json'))
assert sha(O / (BASE + '-Data.json')) == receipt['data_sha256']
for name, digest in receipt['artifacts'].items():
    assert sha(O / name) == digest
assert not data['two_times_target_achieved']
assert not data['strongest_reference_comparison_qualified']


def diagnostic(name, job):
    d = read(O / name)
    proof = O / (job + '-verified.json')
    raw = O / (job + '-evidence.zip')
    assert sha(proof) == d['proof_sha256']
    assert sha(raw) == d['archive_sha256']
    inputs[str(proof)] = sha(proof)
    inputs[str(raw)] = sha(raw)
    assert d['source_runtime_SDK_config_index_CMake_all34objects_baseline_restored']
    assert not d['MTP_805_HTTP_or_performance_qualified']
    return d
