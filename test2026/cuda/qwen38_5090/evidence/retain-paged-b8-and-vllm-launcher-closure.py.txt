sha=lambda raw:hashlib.sha256(raw).hexdigest()

def keep(path,raw):
    assert not path.exists() or path.read_bytes()==raw,path

            gaps=[b['monotonic']-a['monotonic'] for a,b in zip(samples,samples[1:])]
            m=r['sampled_memory']
            assert r['child_exit_code']==0 and r['gpu']['uuid']=='GPU-265ee66c-9b79-cd3b-0c63-dd6942ddb5e6'
            assert len(samples)==m['samples'] and all(g>0 for g in gaps)
            assert max(s['used_bytes'] for s in samples)==m['sampled_peak_adapter_used_bytes']
            assert max(gaps)==m['observed_sample_gap_max_seconds'] and statistics.median(gaps)==m['observed_sample_gap_median_seconds']
            detail['correctness_resources_recomputed']=m
        else:
            assert w['server_exit_code']==127 and not p.get('profiles')
            assert b'No such file or directory' in z.read(base+'wsl-wrapper.log')
            assert not any(n.startswith(base+'client/') for n in z.namelist())
            detail.update(server_exit_code=127,benchmark_profiles=0,model_ready=False,
                          cause='Ordinary launcher omitted from original deployment dependency manifest; no model or benchmark requests')
        for name in z.namelist():
            if name.startswith(base) and name.endswith(('commands.log','commands.jsonl','workflow.json','resource-monitor.json','wsl-wrapper.log')):
                keep(e/(job+'-'+name.removeprefix(base).replace('/','-')),z.read(name))
    jobs[job]=detail
    for suffix in ('-verified.json','-remote-verified.json','-local-replay.log'):
        keep(e/(job+suffix),(out/(job+suffix)).read_bytes())
