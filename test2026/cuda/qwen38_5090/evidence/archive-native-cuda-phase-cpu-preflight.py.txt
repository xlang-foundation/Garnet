from pathlib import Path
import hashlib,json,zipfile

root=Path('D:/CantorAI/5090-study-20261023')
names=['native-cuda-phase-pipe-smoke-attempt1','native-cuda-phase-pipe-smoke-attempt2','native-cuda-phase-profiled-driver-smoke-attempt1']
reports={name:json.loads((root/'results'/name/'report.json').read_bytes()) for name in names}
assert all(reports[name]['status']=='failed' and reports[name]['gpu_model_executed'] is False for name in names[:2])
assert reports[names[-1]]['status']=='profiled_cpu_driver_ipc_verified' and reports[names[-1]]['gpu_model_executed'] is False
