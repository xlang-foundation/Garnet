cmake_minimum_required(VERSION 3.18)
project(garnet_media_image LANGUAGES CXX)
include(../media_common/standalone.cmake)
add_library(garnet_media_image SHARED image_codec.cpp)
if(WIN32)
    set(GARNET_ZLIB_ROOT "" CACHE PATH "zlib include/lib SDK")
    target_include_directories(garnet_media_image PRIVATE "${GARNET_ZLIB_ROOT}/include")
    target_link_libraries(garnet_media_image PRIVATE "${GARNET_ZLIB_ROOT}/lib/zlib.lib" windowscodecs ole32 oleaut32)
else()
    find_package(ZLIB REQUIRED)
    find_package(JPEG REQUIRED)
    find_package(PNG REQUIRED)
    target_link_libraries(garnet_media_image PRIVATE ZLIB::ZLIB JPEG::JPEG PNG::PNG)
endif()
