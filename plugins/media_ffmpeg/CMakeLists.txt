cmake_minimum_required(VERSION 3.18)
project(garnet_media_ffmpeg LANGUAGES CXX)
include(../media_common/standalone.cmake)
add_library(garnet_media_ffmpeg SHARED video_encoder.cpp audio_decoder.cpp)
if(WIN32)
    set(GARNET_FFMPEG_ROOT "" CACHE PATH "Private FFmpeg include/lib SDK")
    target_include_directories(garnet_media_ffmpeg PRIVATE "${GARNET_FFMPEG_ROOT}/include")
    foreach(library avcodec avformat avutil swscale swresample vpl libssl libcrypto zlib)
        target_link_libraries(garnet_media_ffmpeg PRIVATE "${GARNET_FFMPEG_ROOT}/lib/${library}.lib")
    endforeach()
    target_link_libraries(garnet_media_ffmpeg PRIVATE ws2_32 winmm iphlpapi advapi32 bcrypt crypt32 mf mfplat mfuuid mfreadwrite ole32 oleaut32 uuid strmiids secur32 shlwapi vfw32)
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavcodec libavformat libavutil libswscale libswresample)
    target_link_libraries(garnet_media_ffmpeg PRIVATE PkgConfig::FFMPEG)
endif()
